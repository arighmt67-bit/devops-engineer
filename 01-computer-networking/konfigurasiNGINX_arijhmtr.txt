# /etc/nginx/sites-available/default
# NGINX reverse proxy for Node.js + rate limiting (Dicoding submission)

limit_req_zone $binary_remote_addr zone=mylimit:10m rate=6r/m;

server {
    listen 80 default_server;
    listen [::]:80 default_server;

    server_name localhost;

    # Optional: static root (not used when proxying)
    root /var/www/html;
    index index.html index.htm;

    location / {
        proxy_pass http://localhost:8000;

        # HTTP/1.1 & WebSocket headers (safe defaults for Node.js apps)
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;

        # Rate limiting: 6 requests per minute per IP (1 request / 10s)
        limit_req zone=mylimit;
    }
}

server {
    listen 3000;
    listen [::]:3000;

    server_name localhost;

    location / {
        proxy_pass http://localhost:8000;

        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;

        limit_req zone=mylimit;
    }
}
